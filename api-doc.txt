РЕЦЕПТЫ:
1. GET /api/recipes/search?q="..." {  // получить список рецептов(быстрый короткий поиск по неполному совпадению имени)
	request:
		q: string (название рецепта, не обязательно полное(хоть одна буква), забей в нейронку про debounce - реализуй поиск рецептов выпадающим списком)
	responce:
		body:json: {
			object {
				id: number
				name: string,
				macronutrients: { // КБЖУ общее в блюде
					calroies: number, // (в граммах)
					protein: number,
					carbs: number,
					fats: number
				}
			}[] (массивом объектов прилетает, из общего списка рецептов)
		}
}

2. GET /api/recipes/:id {  // получить полную инфу по конкретному рецепту (типа кликнули в первом выпадающем списке, зашли на страничку рецепта)
	request:
		id: number // айди полученное из 1
	responce:
		body:json: {
			id: number
			name: string,
			description: string,
			macronutrients: { // КБЖУ общее в блюде
				calroies: number, // (в граммах)
				protein: number,
				carbs: number,
				fats: number
				},
			ingredients: object { 
				id: number, // id ингр
				name: number,
				amountGr: number; // необходимое для блюда кол-во ингредиента (обяз граммы)
    				customUnit: string | null; // кастом мера для удобства и красоты (пример: штуки, щепотки, чайные, столовые ложки итд \ может быть null, если таковой нет)
    				customAmount: number | null; // количество (в кастом мере)
				} [] // список ингредиентов блюда
			}
		}
	except:
		http-code: 404
		body:json: {
			message: str (думаю что ток "не найдено" прилетать будет, если несущ айди вписать)
		}
			
}

3. GET /api/recipes/searchByIngredients { // поиск рецептов по айдишникам их ингредиентов(фича подбора че сварить)
	request:
		body:json: {
			ids: number[] // массив айдишек выбранных ингредиентов
		}
	responce:
		body:json: {
			object {
				id: number
				name: string,
				macronutrients: {
					calroies: number, // (в граммах)
					protein: number,
					carbs: number,
					fats: number
				}
			}[] (массивом объектов прилетает, из общего списка рецептов)
		}
}
